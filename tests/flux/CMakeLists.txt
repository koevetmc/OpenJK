add_executable(flux_tests flux_tests.c "${MPDir}/game/bg_flux.c")
target_include_directories(flux_tests PRIVATE "${MPDir}/game")
if(MSVC)
	target_compile_options(flux_tests PRIVATE /W4 /WX)
else()
	target_compile_options(flux_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
	target_link_libraries(flux_tests m)
endif()
add_test(NAME flux_physics COMMAND flux_tests)

if(BuildMPGame)
	get_target_property(FluxGameSources ${MPGame} SOURCES)
	add_executable(flux_game_tests flux_game_tests.c ${FluxGameSources})
	target_include_directories(flux_game_tests PRIVATE "${MPDir}/game" $<TARGET_PROPERTY:${MPGame},INCLUDE_DIRECTORIES>)
	target_compile_definitions(flux_game_tests PRIVATE $<TARGET_PROPERTY:${MPGame},COMPILE_DEFINITIONS>)
	if(WIN32)
		target_link_libraries(flux_game_tests winmm)
	else()
		target_link_libraries(flux_game_tests m)
	endif()
	set_target_properties(flux_game_tests PROPERTIES COMPILE_WARNING_AS_ERROR ON)
	add_test(NAME flux_game_integration COMMAND flux_game_tests)
endif()
